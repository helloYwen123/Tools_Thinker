from object_detector import Object_Detector_Tool

def count_cows(image_path, labels=["cow", "bull"], save_image=True):
    detector_tool = Object_Detector_Tool()
    try:
        results = detector_tool.execute(image_path=image_path, labels=labels, threshold=0.85, model_size='tiny', save_object=save_image)
        label_to_labels = detector_tool.output_types['output_results'][0]
        detection_results = {label_to_label: results for label_to_label, results in label_to_labels.items()}
        cow_count = sum(1 for cows in detection_results.values() if 'cow' in cows)
        return cow_count
    except Exception as e:
        print(f"An error occurred while counting cows: {e}")
        return 0

final_result = count_cows("/home/stud/wxie/BLINK_Dataset/Counting/val/images/val_Counting_63_image_1.jpg")

import os
from object_detector import Object_Detector_Tool

# Read image path from environment variable
image_path = os.getenv('image_path')

# Initialize the object detector tool
tool = Object_Detector_Tool()

def classify_objects(image_path):
    # Load the image from the file
    loaded_image = tool.image_objectDetector(image_path, 'baseball')

    counts = tool.objectCount(loaded_image['boxes'], loaded_image['confidence_scores'], 'baseball')

    return counts

def main():
    image_path = 'val_Counting_116_image_1.jpg'
    counts = classify_objects(image_path)

    # Calculate average counts
    average_counts = counts / len(counts)

    final_result = average_counts
    print('The average number of counts is:', final_result)

    if final_result == max(counts):
        print('The image has one ' + str(final_result - 1) + ' most common class in the average counts.')
    else:
        print('The image has two ' + str(round(counts, 2) - 1) + ' most common classes in the average counts.')
    
if __name__ == '__main__':
    main()

from object_detector import Object_Detector_Tool

def count_words_in_sign(file_path):
    tool = Object_Detector_Tool(file_path)
    result = tool.execute(labels=["i"], threshold=0.45, model_size="tiny", save_object=True, saved_image_path="detected_objects")["output_types"][0][1]
    final_result = {}
    for label, data in result.items():
        final_result[label] = data["count"]
    return final_result
